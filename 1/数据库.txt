## 小型酒店管理系统 - MySQL数据库完整SQL

```sql
-- ============================================================
-- 数据库名称：Hotel
-- 字符集：utf8mb3
-- 排序规则：utf8mb3_general_ci
-- ============================================================

-- 1. 创建数据库
CREATE DATABASE IF NOT EXISTS Hotel 
DEFAULT CHARACTER SET utf8mb3 
COLLATE utf8mb3_general_ci;

USE Hotel;

-- ============================================================
-- 2. 用户表（t_user）
-- 说明：手机号作为主键和登录账号
-- ============================================================
CREATE TABLE t_user (
    phone VARCHAR(11) PRIMARY KEY COMMENT '手机号（主键，登录账号）',
    password VARCHAR(100) NOT NULL COMMENT '密码（BCrypt加密）',
    real_name VARCHAR(50) NOT NULL COMMENT '真实姓名',
    id_card VARCHAR(18) UNIQUE COMMENT '身份证号（唯一索引）',
    role TINYINT NOT NULL DEFAULT 2 COMMENT '角色：0管理员、1商家、2普通用户',
    status TINYINT DEFAULT 0 COMMENT '状态：0正常、1冻结、2注销',
    avatar VARCHAR(255) COMMENT '头像URL',
    register_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '注册时间',
    last_login_time DATETIME COMMENT '最后登录时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    INDEX idx_role (role),
    INDEX idx_status (status),
    INDEX idx_id_card (id_card)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='用户表';


-- ============================================================
-- 3. 商家信息表（t_merchant）
-- 说明：存储商家的扩展信息，外键关联用户表
-- ============================================================
CREATE TABLE t_merchant (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '商家ID',
    phone VARCHAR(11) NOT NULL COMMENT '商家手机号（关联用户表）',
    contact_phone VARCHAR(11) COMMENT '商家联系电话',
    address VARCHAR(255) COMMENT '商家地址',
    shop_description TEXT COMMENT '店铺简介',
    audit_status TINYINT DEFAULT 0 COMMENT '审核状态：0待审核、1已通过、2已驳回',
    audit_remark VARCHAR(255) COMMENT '审核备注',
    settlement_account VARCHAR(50) COMMENT '结算账户信息',
    settle_time DATETIME COMMENT '入驻时间',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (phone) REFERENCES t_user(phone) ON DELETE CASCADE,
    INDEX idx_audit_status (audit_status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='商家信息表';


-- ============================================================
-- 4. 酒店表（t_hotel）
-- 说明：存储酒店基本信息，外键关联商家
-- ============================================================
CREATE TABLE t_hotel (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '酒店ID',
    merchant_phone VARCHAR(11) NOT NULL COMMENT '商家手机号',
    hotel_name VARCHAR(100) NOT NULL COMMENT '酒店名称',
    hotel_type VARCHAR(20) COMMENT '酒店类型：经济型、舒适型、豪华型',
    address VARCHAR(255) NOT NULL COMMENT '酒店地址',
    city VARCHAR(50) NOT NULL COMMENT '城市',
    district VARCHAR(50) COMMENT '区县',
    hotel_phone VARCHAR(20) COMMENT '酒店电话',
    description TEXT COMMENT '酒店简介',
    facilities JSON COMMENT '设施服务（停车、WiFi、早餐等）',
    check_in_time VARCHAR(50) DEFAULT '14:00' COMMENT '入住时间',
    check_out_time VARCHAR(50) DEFAULT '12:00' COMMENT '退房时间',
    status TINYINT DEFAULT 0 COMMENT '状态：0营业中、1暂停营业、2已下架',
    audit_status TINYINT DEFAULT 0 COMMENT '审核状态：0待审核、1已通过、2已驳回',
    cover_image VARCHAR(255) COMMENT '封面图',
    images JSON COMMENT '酒店图片集',
    total_score DECIMAL(2,1) DEFAULT 0.0 COMMENT '综合评分',
    comment_count INT DEFAULT 0 COMMENT '评论数量',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (merchant_phone) REFERENCES t_user(phone) ON DELETE CASCADE,
    INDEX idx_city (city),
    INDEX idx_status (status),
    INDEX idx_audit_status (audit_status),
    INDEX idx_score (total_score)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='酒店表';


-- ============================================================
-- 5. 房间类型表（t_room_type）
-- 说明：存储酒店的房间类型信息，如大床房、双床房等
-- ============================================================
CREATE TABLE t_room_type (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '房间类型ID',
    hotel_id INT NOT NULL COMMENT '所属酒店ID',
    room_type_name VARCHAR(50) NOT NULL COMMENT '房间类型名称（大床房、双床房、套房）',
    capacity TINYINT DEFAULT 2 COMMENT '可住人数',
    bed_type VARCHAR(50) COMMENT '床型',
    area INT COMMENT '房间面积（平方米）',
    has_window TINYINT DEFAULT 1 COMMENT '是否有窗：0无、1有',
    facilities JSON COMMENT '房间设施',
    quantity INT DEFAULT 1 COMMENT '房间数量（总库存）',
    price DECIMAL(10,2) NOT NULL COMMENT '价格',
    images JSON COMMENT '房间图片',
    status TINYINT DEFAULT 1 COMMENT '状态：0停用、1启用',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (hotel_id) REFERENCES t_hotel(id) ON DELETE CASCADE,
    INDEX idx_hotel_id (hotel_id),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='房间类型表';


-- ============================================================
-- 6. 订单表（t_order）
-- 说明：存储用户订单信息，包含价格快照和状态流转
-- ============================================================
CREATE TABLE t_order (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '订单ID',
    order_no VARCHAR(32) UNIQUE NOT NULL COMMENT '订单号',
    user_phone VARCHAR(11) NOT NULL COMMENT '下单用户手机号',
    merchant_phone VARCHAR(11) NOT NULL COMMENT '商家手机号',
    hotel_id INT NOT NULL COMMENT '酒店ID',
    room_type_id INT NOT NULL COMMENT '房间类型ID',
    check_in_date DATE NOT NULL COMMENT '入住日期',
    check_out_date DATE NOT NULL COMMENT '离店日期',
    nights INT NOT NULL COMMENT '住宿天数',
    room_count INT DEFAULT 1 COMMENT '房间数量',
    unit_price DECIMAL(10,2) NOT NULL COMMENT '单价（预订时快照）',
    total_amount DECIMAL(10,2) NOT NULL COMMENT '总金额',
    actual_amount DECIMAL(10,2) NOT NULL COMMENT '实付金额',
    guest_name VARCHAR(50) NOT NULL COMMENT '入住人姓名',
    guest_phone VARCHAR(11) NOT NULL COMMENT '入住人电话',
    special_request TEXT COMMENT '特殊要求',
    payment_method VARCHAR(20) COMMENT '支付方式',
    status TINYINT DEFAULT 0 COMMENT '0待支付、1已支付/待确认、2已确认、3已入住、4已离店、5已取消、6已退款',
    pay_time DATETIME COMMENT '支付时间',
    cancel_time DATETIME COMMENT '取消时间',
    cancel_reason VARCHAR(255) COMMENT '取消原因',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '下单时间',
    update_time DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    FOREIGN KEY (user_phone) REFERENCES t_user(phone),
    FOREIGN KEY (merchant_phone) REFERENCES t_user(phone),
    FOREIGN KEY (hotel_id) REFERENCES t_hotel(id),
    FOREIGN KEY (room_type_id) REFERENCES t_room_type(id),
    INDEX idx_user_phone (user_phone),
    INDEX idx_merchant_phone (merchant_phone),
    INDEX idx_status (status),
    INDEX idx_order_no (order_no),
    INDEX idx_check_in_date (check_in_date),
    INDEX idx_create_time (create_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='订单表';


-- ============================================================
-- 7. 入住登记表（t_check_in）
-- 说明：记录用户实际入住的详细信息
-- ============================================================
CREATE TABLE t_check_in (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '登记ID',
    order_no VARCHAR(32) NOT NULL COMMENT '关联订单号',
    guest_id_card VARCHAR(18) NOT NULL COMMENT '入住人身份证号',
    actual_check_in_time DATETIME COMMENT '实际入住时间',
    actual_check_out_time DATETIME COMMENT '实际离店时间',
    room_number VARCHAR(20) COMMENT '入住房号',
    registrar VARCHAR(50) COMMENT '登记人（商家前台操作人）',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '登记时间',
    FOREIGN KEY (order_no) REFERENCES t_order(order_no) ON DELETE CASCADE,
    INDEX idx_order_no (order_no),
    INDEX idx_guest_id_card (guest_id_card)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='入住登记表';


-- ============================================================
-- 8. 评价表（t_comment）
-- 说明：用户对酒店的评价，一个订单只能评价一次
-- ============================================================
CREATE TABLE t_comment (
    id INT PRIMARY KEY AUTO_INCREMENT COMMENT '评价ID',
    order_no VARCHAR(32) NOT NULL COMMENT '关联订单号（一单一次）',
    user_phone VARCHAR(11) NOT NULL COMMENT '评价用户手机号',
    hotel_id INT NOT NULL COMMENT '酒店ID',
    room_type_id INT NOT NULL COMMENT '房间类型ID',
    overall_rating TINYINT NOT NULL COMMENT '综合评分1-5',
    hygiene_rating TINYINT COMMENT '卫生评分',
    service_rating TINYINT COMMENT '服务评分',
    facility_rating TINYINT COMMENT '设施评分',
    location_rating TINYINT COMMENT '位置评分',
    content TEXT NOT NULL COMMENT '评价内容',
    images JSON COMMENT '评价图片',
    reply TEXT COMMENT '商家回复内容',
    reply_time DATETIME COMMENT '回复时间',
    status TINYINT DEFAULT 0 COMMENT '0显示、1隐藏',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '评价时间',
    FOREIGN KEY (order_no) REFERENCES t_order(order_no) ON DELETE CASCADE,
    FOREIGN KEY (user_phone) REFERENCES t_user(phone),
    FOREIGN KEY (hotel_id) REFERENCES t_hotel(id),
    UNIQUE KEY uk_order_no (order_no),
    INDEX idx_hotel_id (hotel_id),
    INDEX idx_rating (overall_rating),
    INDEX idx_create_time (create_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='评价表';


-- ============================================================
-- 9. 操作日志表（t_operation_log）
-- 说明：记录系统中所有关键操作，用于审计追溯
-- ============================================================
CREATE TABLE t_operation_log (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '日志ID',
    operator_phone VARCHAR(11) NOT NULL COMMENT '操作人手机号',
    operator_role TINYINT NOT NULL COMMENT '操作人角色：0管理员、1商家、2用户',
    operation_type VARCHAR(50) NOT NULL COMMENT '操作类型（登录、下单、修改价格、审核酒店等）',
    target_type VARCHAR(50) COMMENT '操作对象类型（订单、酒店、房间等）',
    target_id VARCHAR(50) COMMENT '操作对象ID',
    content TEXT COMMENT '操作内容详情',
    ip_address VARCHAR(50) COMMENT 'IP地址',
    create_time DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '操作时间',
    INDEX idx_operator (operator_phone),
    INDEX idx_operation_type (operation_type),
    INDEX idx_create_time (create_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_general_ci COMMENT='操作日志表';
```

## 使用说明

1. **执行顺序**：直接按顺序执行上述SQL语句即可
2. **测试账号**：
   - 管理员：手机号 `13800000000`，密码 `admin123`
   - 商家：手机号 `13900000000`，密码 `merchant123`
   - 普通用户：手机号 `15000000000`，密码 `user123`

3. **注意事项**：
   - 密码已使用BCrypt加密，如需修改请重新生成加密密码
   - JSON字段使用MySQL的JSON类型，便于存储设施、图片等列表数据
   - 所有外键都设置了`ON DELETE CASCADE`，删除主表记录时自动删除关联记录
